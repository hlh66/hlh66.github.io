⌨️ TCP 常用命令速查

# 查看TCP连接状态 (Linux/Mac)
netstat -tlnp
ss -tlnp

# 查看指定端口占用
lsof -i :8080
netstat -tlnp | grep 8080

# TCP端口连通性测试
nc -zv host port
telnet host port

# Windows 查看端口
netstat -ano | findstr :8080

# 测试TCP连接 (Python)
python -c "import socket; s=socket.socket(); s.settimeout(3); s.connect(('host',port)); print('OPEN'); s.close()"

# curl 测试TCP
curl -v telnet://host:port
